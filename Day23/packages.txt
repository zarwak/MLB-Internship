# Streamlit Cloud runs `apt-get install` on every line in this file
# during the build - the ONLY way to get Tesseract onto the deployed
# container, since pytesseract is just a Python wrapper around a native
# binary that pip cannot install (see HOW_TO_RUN.txt section 0b for the
# same fact on Windows, where it needs a separate installer instead).
# Same folder rule as requirements.txt: Streamlit Cloud only reads this
# from the repo root or the app's own folder, hence the duplicate copy
# in mini_project/.
tesseract-ocr

# easyocr (and opencv, which paddleocr/doctr also pull in transitively)
# imports cv2 internally even though our own code never does - `import
# cv2` then fails with "libGL.so.1: cannot open shared object file"
# because Streamlit Cloud's container has none of the system graphics
# libraries OpenCV expects, only being a headless Debian image. libgl1 +
# libglib2.0-0 alone (what fixed this in the DocProcessor project) was
# not the complete set here - this is the fuller list commonly needed
# for opencv-python (non-headless) on a minimal Debian image.
libgl1
libglib2.0-0
libsm6
libxext6
libxrender1
libgomp1
